<script>
const tokenKey = 'galaxy_access_token';
const roleKey = 'galaxy_role';

function escapeHtml(value) {
    return String(value ?? '')
        .replaceAll('&', '&amp;')
        .replaceAll('<', '&lt;')
        .replaceAll('>', '&gt;')
        .replaceAll('"', '&quot;')
        .replaceAll("'", '&#039;');
}

function showLogin(message = '') {
    document.getElementById('loginView').classList.remove('hidden');
    document.getElementById('dashboardView').classList.add('hidden');
    document.getElementById('loginError').textContent = message;
}

function showDashboard() {
    document.getElementById('loginView').classList.add('hidden');
    document.getElementById('dashboardView').classList.remove('hidden');
}

async function api(path, options = {}) {
    const token = localStorage.getItem(tokenKey);
    const headers = { ...(options.headers || {}) };
    if (token) headers['Authorization'] = `Bearer ${token}`;
    const response = await fetch(path, { ...options, headers });
    if (response.status === 401) {
        localStorage.removeItem(tokenKey);
        localStorage.removeItem(roleKey);
        showLogin('Sesión expirada. Vuelve a iniciar sesión.');
        throw new Error('unauthorized');
    }
    const data = await response.json().catch(() => ({}));
    if (!response.ok) throw new Error(data.detail || `HTTP ${response.status}`);
    return data;
}

async function loadDashboard() {
    document.getElementById('dashboardError').textContent = '';
    try {
        const me = await api('/api/v1/auth/me');
        const user = me.user;
        if (!['OWNER', 'ADMIN'].includes(user.role)) {
            throw new Error('Tu usuario no tiene permisos para el Dashboard.');
        }
        const data = await api('/api/v1/dashboard/overview');

        document.getElementById('businessName').textContent = data.business?.name || 'Galaxy';
        document.getElementById('businessMeta').textContent = `${data.business?.industry || ''} · ${data.business?.currency || ''} · ${user.full_name || user.email}`;
        document.getElementById('roleBadge').textContent = user.role;
        document.getElementById('businessId').textContent = data.business?.id || user.business_id;

        const a = data.analytics || {};
        document.getElementById('sales').textContent = `$${Number(a.orders?.sales_total || 0).toFixed(2)}`;
        document.getElementById('orders').textContent = Number(a.orders?.total || 0);
        document.getElementById('conversations').textContent = Number(a.conversations_total || 0);
        document.getElementById('lowStock').textContent = Number(a.inventory?.low_stock || 0);
        document.getElementById('handoffs').textContent = Number(a.handoffs_total || 0);
        document.getElementById('audits').textContent = Number(a.audits_today || 0);
        document.getElementById('inventoryMeta').textContent = `Registros: ${Number(a.inventory?.records || 0)} · Fuera de stock: ${Number(a.inventory?.out_of_stock || 0)}`;

        const insights = document.getElementById('insights');
        insights.innerHTML = '';
        (data.insights || []).forEach(item => {
            insights.innerHTML += `<li><strong>${escapeHtml(item.title || 'Insight')}</strong><br>${escapeHtml(item.description || '')}<br><span class="muted">Prioridad: ${escapeHtml(item.priority || 'normal')} · Estado: ${escapeHtml(item.status || '')}</span></li>`;
        });
        if (!insights.children.length) insights.innerHTML = '<li class="muted">No hay insights abiertos.</li>';

        const recs = document.getElementById('recommendations');
        recs.innerHTML = '';
        (data.recommendations || []).forEach(item => {
            recs.innerHTML += `<li><strong>${escapeHtml(item.title || 'Recomendación')}</strong><br>${escapeHtml(item.description || '')}<br><span class="muted">Prioridad: ${escapeHtml(item.priority || 'normal')} · Aprobación: ${item.requires_approval ? 'sí' : 'no'}</span></li>`;
        });
        if (!recs.children.length) recs.innerHTML = '<li class="muted">No hay recomendaciones pendientes.</li>';

        const products = document.getElementById('products');
        products.innerHTML = '';
        (data.top_products || []).forEach(item => {
            products.innerHTML += `<li><strong>${escapeHtml(item.product_name || 'Producto')}</strong><br>${Number(item.units || 0)} unidades · $${Number(item.revenue || 0).toFixed(2)}</li>`;
        });
        if (!products.children.length) products.innerHTML = '<li class="muted">Todavía no hay ventas registradas.</li>';
        showDashboard();
    } catch (err) {
        if (err.message !== 'unauthorized') document.getElementById('dashboardError').textContent = err.message;
    }
}

document.getElementById('loginForm').addEventListener('submit', async (event) => {
    event.preventDefault();
    const error = document.getElementById('loginError');
    error.textContent = '';
    const body = new URLSearchParams();
    body.set('username', document.getElementById('email').value.trim());
    body.set('password', document.getElementById('password').value);
    try {
        const response = await fetch('/api/v1/auth/login', {
            method: 'POST',
            headers: {'Content-Type': 'application/x-www-form-urlencoded'},
            body,
        });
        const data = await response.json();
        if (!response.ok) throw new Error(data.detail || 'No se pudo iniciar sesión.');
        if (!['OWNER', 'ADMIN'].includes(data.role)) throw new Error('El usuario no tiene acceso al Dashboard.');
        localStorage.setItem(tokenKey, data.access_token);
        localStorage.setItem(roleKey, data.role);
        await loadDashboard();
    } catch (err) {
        error.textContent = err.message;
    }
});

document.getElementById('refreshBtn').addEventListener('click', loadDashboard);
document.getElementById('logoutBtn').addEventListener('click', () => {
    localStorage.removeItem(tokenKey);
    localStorage.removeItem(roleKey);
    showLogin('Sesión cerrada.');
});

if (localStorage.getItem(tokenKey)) loadDashboard();
else showLogin();
